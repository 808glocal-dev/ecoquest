// 아래 match 블록만 현재 운영 firestore.rules의 service cloud.firestore / match /databases/{database}/documents 안에 병합하세요.
// 기존 rules 전체를 이 파일로 교체하지 마세요.
// 화면 코드가 회원 데이터를 보호하지는 않습니다. 실제 프로덕션 투입 전에 Firebase 에뮬레이터로 반드시 검증하세요.
match /planetPresence/{planetId}/players/{playerUid} {
  function validPlanet() {
    return planetId in ['forest','ocean','wind','home','food','sharing','quiet'];
  }
  function validPresence() {
    let d = request.resource.data;
    return d.keys().hasOnly(['name','x','z','updatedAt','shareProgress','count','avatar'])
      && d.keys().hasAll(['name','x','z','updatedAt','shareProgress','count','avatar'])
      && d.name is string && d.name.size() <= 20
      && d.avatar == 'bunny'
      && d.x is number && d.x >= -28 && d.x <= 28
      && d.z is number && d.z >= -28 && d.z <= 28
      && d.updatedAt == request.time
      && d.shareProgress is bool
      && d.count is int && d.count >= 0 && d.count <= 100000
      && (d.shareProgress || d.count == 0);
  }
  // 인증 회원에게 공개되는 게임 속 위치만 조회. 현실 위치나 이메일은 저장하지 않습니다.
  allow read: if request.auth != null && validPlanet();
  allow create, update: if request.auth != null && request.auth.uid == playerUid
    && validPlanet() && validPresence();
  allow delete: if request.auth != null && request.auth.uid == playerUid && validPlanet();
}
